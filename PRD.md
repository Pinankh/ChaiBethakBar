service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }

    match /outlet_config/{docId} {
      allow read, write: if isSignedIn();
    }
    match /tables/{tableId} {
      allow read, write: if isSignedIn();
    }
    match /menu_items/{itemId} {
      allow read, write: if isSignedIn();
    }
    match /orders/{orderId} {
      allow read, create, update: if isSignedIn();
      allow delete: if false;
    }
  }
}

Tax & Calculation SpecificationlineSubtotal = unitPricePaise * qty for each item; subtotalPaise = sum(lineSubtotal).Add parcelChargePaise (only for TAKEAWAY when enabled; 0% extra tax on parcel charge so ₹10 stays ₹10).Subtract discountPaise (clamped between 0L and subtotalPaise + parcelChargePaise).When calculating tax, distribute item discounts proportionally (discountedLine = Math.round(lineSubtotal * (subtotalPaise - min(discountPaise, subtotalPaise)) / subtotalPaise.toDouble()) when subtotalPaise > 0).For items with taxType == "GST": sum Math.round(discountedLine * taxPercent / 100.0) into totalGst; split cgstPaise = totalGst / 2, sgstPaise = totalGst - cgstPaise.For items with taxType == "VAT": sum Math.round(discountedLine * taxPercent / 100.0) into vatPaise.For items with taxType == "NONE" (or taxPercent == 0): tax is 0L.rawTotal = subtotalPaise + parcelChargePaise - discountPaise + cgstPaise + sgstPaise + vatPaise.Nearest-₹1 rounding: roundedTotal = ((rawTotal + 50L) / 100L) * 100L; roundOffPaise = roundedTotal - rawTotal; totalPaise = roundedTotal.6.
